# === Alias mapping (shared Excalibur/React secrets → Posts EMDR) ===

# MAX_CHAT_ID              ← MAX_CHANNEL_CHAT_ID | EXCALIBUR_MAX_CHANNEL_CHAT_ID
#   (НЕ MAX_NOTIFY_CHAT_ID — это ЛС Отчётика, не канал)
# TELEGRAM_BOT_TOKEN       ← EXCALIBUR_TELEGRAM_BOT_TOKEN
# TELEGRAM_CHANNEL_CHAT_IDS ← valid TELEGRAM_CHANNEL_CHAT_IDS | valid EXCALIBUR_TELEGRAM_CHANNEL_CHAT_IDS
#   если список содержит снятые каналы (@morozova_emdr, @natalia_morozova_psy) — не используется
#   default: @nmorozova_emdr
# TELEGRAM_CHANNEL_UTM_SOURCES ← EXCALIBUR_TELEGRAM_CHANNEL_UTM_SOURCES (default tg1)
# FTP_*                    ← REACT_FTP_* если FTP_* пустые
# WORDPRESS_URL            ← WP_HOME | WP_SITE_URL | PUBLIC_SITE_URL
# WORDPRESS_USER           ← WP_USER | WP_ADMIN_USER
# WORDPRESS_APP_PASSWORD   ← WP_APP_PASSWORD
# WORDPRESS_SITE_URL       ← PUBLIC_SITE_URL | WP_HOME | WP_SITE_URL
# VPS_WEBHOOK_SECRET       ← EXCALIBUR_VPS_WEBHOOK_SECRET
#
